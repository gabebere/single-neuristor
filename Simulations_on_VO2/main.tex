\begin{frame}{Starting Point: Electrical Dynamics}

\begin{columns}[T,onlytextwidth]
\begin{column}{0.59\textwidth}

\textbf{Yuanhang voltage-source node equation}

\vspace{0.18cm}

\[
C\frac{dV}{dt}
=
\frac{V_{in}-V}{R_{series}}
-
\frac{V}{R_{\mathrm{VO_2}}}
\]

\vspace{0.22cm}

\begin{itemize}
    \item \(C\,dV/dt\): charge stored at the neuristor node.
    \item \((V_{in}-V)/R_{series}\): current delivered by the source.
    \item \(V/R_{\mathrm{VO_2}}\): nonlinear leakage through VO\textsubscript{2}.
    \item \(R_{\mathrm{VO_2}}=R(T,\mathcal H)\): temperature and memory couple the electrical and thermal problems.
\end{itemize}

\end{column}

\begin{column}{0.36\textwidth}
\centering
\includegraphics[width=0.92\linewidth]{../vo2_device.png}

\vspace{0.2cm}
{\small Voltage-source equivalent circuit.}
\end{column}
\end{columns}

\slidesource{Starting point: Y.-H. Zhang et al., Nature Communications (2024).}
\note{Explain the competition: the load charges the capacitor, while the VO2 path discharges it. The resistance is not fixed; it is controlled by the thermal state.}
\end{frame}

\begin{frame}{The Feedback Mechanism: Thermal Dynamics}

\begin{columns}[T,onlytextwidth]
\begin{column}{0.61\textwidth}

\textbf{Single-neuristor reduction}

\vspace{0.15cm}

\[
C_{th}\frac{dT}{dt}
=
\underbrace{\frac{V^2}{R_{\mathrm{VO_2}}}}_{\text{Joule heating}}
-
\underbrace{S_e(T-T_0)}_{\text{cooling}}
+
\underbrace{\sigma\eta(t)}_{\text{optional noise}}
\]

\vspace{0.2cm}

\begin{itemize}
    \item \(C_{th}\): thermal inertia of the active device.
    \item \(S_e\): thermal conductance to substrate/environment.
    \item Heating lowers \(R_{\mathrm{VO_2}}\); lower resistance changes the voltage and power.
    \item This delayed negative feedback enables relaxation oscillations.
\end{itemize}

\end{column}

\begin{column}{0.34\textwidth}
\begin{block}{Two coupled state variables}
\[
V(t),\qquad T(t)
\]
\end{block}

\begin{block}{Material memory}
\[
\mathcal H=
\{\delta,T_r,g_r,T_{pr},\ldots\}
\]
\end{block}

\begin{center}
\small
Electrical storage\\
\(\Updownarrow\)\\
Joule heating\\
\(\Updownarrow\)\\
Hysteretic switching
\end{center}
\end{column}
\end{columns}

\slidesource{For one isolated device, the inter-neuristor thermal coupling term is zero.}
\note{Keep this physical: electrical and thermal storage provide two timescales. Hysteresis supplies distinct heating and cooling thresholds.}
\end{frame}

\begin{frame}{Material Law: Resistance and Hysteresis}

\begin{columns}[T,onlytextwidth]
\begin{column}{0.45\textwidth}
\small

\[
R(T,\mathcal H)
=
R_0 e^{E_a/(k_BT)}
\underbrace{g(T,\mathcal H)}_{\text{insulating fraction}}
+R_m
\]

\[
g=
\frac{1}{2}
+\frac{1}{2}\tanh
\left[
\beta\left(
\delta\frac{w}{2}+T_c-(T+T_p)
\right)
\right]
\]

\scriptsize
\begin{itemize}
    \setlength\itemsep{0.06cm}
    \item \(T_c,w,\beta\): center, width, and steepness.
    \item \(\delta=+1\): heating; \(\delta=-1\): cooling.
    \item \(T_p\): minor-loop shift after a reversal.
    \item The same law is used in both source configurations.
\end{itemize}

\end{column}

\begin{column}{0.52\textwidth}
\centering
\begin{block}{Insulating and metallic limits}
\small
\(g\approx1\): insulating Arrhenius contribution.\\[0.18cm]
\(g\approx0\): metallic floor \(R\approx R_m\).
\end{block}
\begin{block}{History matters}
\small
\(T_p=0\) before the first reversal. After a reversal, the stored state shifts the effective temperature to form a minor branch.
\end{block}
\end{column}
\end{columns}

\slidesource{Hysteresis: L. A. L. de Almeida et al., Optical Engineering (2002).}
\note{Point out that fitting in log resistance is essential because the transition spans orders of magnitude. The specimen fit is introduced after the current-source reduction.}
\end{frame}

\begin{frame}{Minor Loops: The Proximity Function}
\begin{columns}[T,onlytextwidth]
\begin{column}{0.58\textwidth}
\footnotesize
\textbf{A reversal shifts the effective temperature}
\[
T_{\mathrm{eff}}=T+T_p,\qquad
T_p=m\,T_{pr}P(x;\gamma),\qquad
x=\frac{T-T_r}{T_{pr}+\varepsilon}.
\]
\[
\boxed{P(x;\gamma)=\frac{1-\sin(\gamma x)}{2}
\left[1+\tanh(\pi^2-2\pi x)\right]}
\]
\begin{itemize}
\setlength\itemsep{0.05cm}
\item \(T_r\): latest detected reversal temperature; \(T_{pr}\): signed shift in K.
\item \(m=0\) before the first reversal; \(m=1\) afterward.
\item Near \(x=0\), \(P\approx1\): the shift preserves the reversal state.
\item As \(x\to+\infty\), \(P\to0\): the selected major branch is recovered.
\end{itemize}
\end{column}

\begin{frame}{Minor Loops: Set the Shift at Each Reversal}
\begin{columns}[T,onlytextwidth]
\begin{column}{0.51\textwidth}
\footnotesize
\textbf{1. Save the state, then change direction}
\[
T_r\leftarrow T,\quad
g_r\leftarrow g(T,\mathcal H^-),\quad
\delta\leftarrow\operatorname{sign}(T-T^{\mathrm{last}}).
\]
\(\mathcal H^-\) is the memory before changing the branch.

\vspace{0.18cm}
\textbf{2. Match the fraction at the reversal}
\[
g_r\simeq\frac12+\frac12\tanh\!\left[
\beta\left(\delta\frac{w}{2}+T_c-T_r-T_{pr}\right)\right].
\]
Inverting the tanh gives the stored shift:
\[
\boxed{T_{pr}=\delta\frac{w}{2}+T_c-T_r
-\frac{\operatorname{arctanh}(2g_r-1)}{\beta}.}
\]
\end{column}
\begin{column}{0.46\textwidth}
\footnotesize
\textbf{3. Follow the resulting minor branch}
\[
g(T,\mathcal H)=\frac12+\frac12\tanh\!\left[
\beta\left(\delta\frac{w}{2}+T_c-T-T_p\right)\right].
\]
\begin{block}{Why this carries memory}
\footnotesize
\(T_r,g_r\) set the starting point; \(T_{pr}P\) carries that state into the new direction. Each new reversal replaces the stored anchor and shift.
\end{block}
\begin{alertblock}{Detection in the solver}
\footnotesize
Accept motion if \(|T-T^{\mathrm{last}}|>0.01\) K. Reverse if its sign differs from \(\delta\). Smaller motion accumulates from the last accepted temperature.
\end{alertblock}
\end{column}
\end{columns}
\slidesource{\texttt{model.py:HysteresisArray}: matching uses \(P(0)\approx1\). Only exact float32 endpoints \(g_r=0,1\) are regularized for arctanh.}
\note{The old branch is evaluated at the detected temperature, before delta changes; g_r is not simply the previous sample's fraction. The equation for Tpr follows by applying inverse tanh to the fraction law. Matching is approximate only because the implemented P(0) is nearly, rather than exactly, one and calculations use float32. The default effective width is w; in configured runs width_factor multiplies w. Tlast advances after accepted motion. The implementation keeps the latest reversal anchor, not a stack of all earlier turning points, so do not claim general return-point memory.}
\end{frame}

\begin{frame}{From Voltage Drive to Ideal Current Drive}

\begin{columns}[T,onlytextwidth]
\begin{column}{0.47\textwidth}
\begin{block}{Original voltage source}
\[
C\frac{dV}{dt}
=
\frac{V_{in}-V}{R_{series}}
-\frac{V}{R}
\]
\[
I_{\mathrm{src}} \text{ changes with } V
\]
\end{block}
\end{column}

\begin{column}{0.47\textwidth}
\begin{alertblock}{Ideal imposed current}
\[
I_{in}=C\frac{dV}{dt}+\frac{V}{R}
\]
\[
\boxed{
C\frac{dV}{dt}=I_{in}-\frac{V}{R}
}
\]
\end{alertblock}
\end{column}
\end{columns}
